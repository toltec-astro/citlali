95auto JincMapmaker::jinc_func(double r, double a, double b, double c, double r_max, double l_d) {
96 if (r!=0) {
97 // unitless radius
98 r = r/l_d;
99 // first jinc function
100 auto jinc_1 = 2.*boost::math::cyl_bessel_j(1,2.*pi*r/a)/(2.*pi*r/a);
101 // exponential
102 auto exp_func = exp(-pow(2.*r/b,c));
103 // second jinc function
104 auto jinc_2 = 2.*boost::math::cyl_bessel_j(1,3.831706*r/r_max)/(3.831706*r/r_max);
105 // jinc1 x exp x jinc2
106 return jinc_1*exp_func*jinc_2;
107 }
108 else {
109 return 1.0;
110 }
111}

113void JincMapmaker::allocate_jinc_matrix(double pixel_size_rad) {
114 l_d[0] = (1.1/1000)/50;
115 l_d[1] = (1.4/1000)/50;
116 l_d[2] = (2.0/1000)/50;
117
118 // loop through lambda/diameters
119 for (const auto &ld: l_d) {
120 // get shape params
121 auto a = shape_params[ld.first](0);
122 auto b = shape_params[ld.first](1);
123 auto c = shape_params[ld.first](2);
124
125 // maximum radius in pixels
126 int r_max_pix = std::floor(r_max*ld.second/pixel_size_rad);
127
128 // pixel centers within max radius
129 Eigen::VectorXd pixels = Eigen::VectorXd::LinSpaced(2*r_max_pix + 1,-r_max_pix, r_max_pix);
130
131 // allocate jinc weights
132 jinc_weights_mat[ld.first].setZero(2*r_max_pix + 1,2*r_max_pix + 1);
133
134 // loop through matrix rows
135 for (Eigen::Index i=0; i<pixels.size(); ++i) {
136 // loop through matrix cols
137 for (Eigen::Index j=0; j<pixels.size(); ++j) {
138 // radius of current pixel in radians
139 double r = pixel_size_rad*sqrt(pow(pixels(i),2) + pow(pixels(j),2));
140 // calculate jinc weight at pixel
141 jinc_weights_mat[ld.first](i,j) = jinc_func(r,a,b,c,r_max,ld.second);
142 }
143 }
144 }

147void JincMapmaker::calculate_jinc_splines() {
148 l_d[0] = (1.1/1000)/50;
149 l_d[1] = (1.4/1000)/50;
150 l_d[2] = (2.0/1000)/50;
151
152 // loop through lambda/diameters
153 for (const auto &ld: l_d) {
154 // get shape params
155 auto a = shape_params[ld.first](0);
156 auto b = shape_params[ld.first](1);
157 auto c = shape_params[ld.first](2);
158
159 // radius vector in radians
160 auto radius = Eigen::VectorXd::LinSpaced(n_pts_splines, 0, r_max*ld.second);
161 // jinc weights on dense vector
162 Eigen::VectorXd jinc_weights(radius.size());
163
164 Eigen::Index j = 0;
165
166 for (const auto &r: radius) {
167 // calculate jinc weights
168 jinc_weights(j) = jinc_func(r,a,b,c,r_max,ld.second);
169 ++j;
170 }
171 // create spline class
172 engine_utils::SplineFunction s;
173 // spline interpolate
174 s.interpolate(radius, jinc_weights);
175 // store jinc spline
176 jinc_splines[ld.first] = s;
177 }

181void JincMapmaker::allocate_pointing(map_buffer_t &mb, double weight, double cos_2angle, double sin_2angle,
182 Eigen::Index map_index, int ir, int ic) {
183 int pix = mb.n_rows * ic + ir;
184 Eigen::MatrixXd& matrix = mb.pointing[map_index];
185
186 // calculate reused expressions
187 double weight_cos_2angle = weight * cos_2angle;
188 double weight_sin_2angle = weight * sin_2angle;
189 double weight_cos2_2angle = weight * std::pow(cos_2angle, 2);
190 double weight_sin2_2angle = weight * std::pow(sin_2angle, 2);
191 double weight_cos_sin_2angle = weight * cos_2angle * sin_2angle;
192
193 // update pointing matrix
194 matrix(pix, 0) += weight;
195 matrix(pix, 1) += weight_cos_2angle;
196 matrix(pix, 2) += weight_sin_2angle;
197 matrix(pix, 3) = matrix(pix, 1); // previously set to += weight*cos_2angle, then directly assigned here
198 matrix(pix, 4) += weight_cos2_2angle;
199 matrix(pix, 5) += weight_cos_sin_2angle;
200 matrix(pix, 6) = matrix(pix, 2); // previously set to += weight*sin_2angle, then directly assigned here
201 matrix(pix, 7) = matrix(pix, 5); // reuse the result from matrix(pix,5)
202 matrix(pix, 8) += weight_sin2_2angle;
203}
