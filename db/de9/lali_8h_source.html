37void Lali::pipeline(KidsProc &kidsproc, RawObs &rawobs) {
38 using tuple_t = TCData<TCDataKind::RTC, Eigen::MatrixXd>;
39
40 // initialize number of completed scans
41 n_scans_done = 0;
42
43 // declare random number generator
44 boost::random::mt19937 eng;
45
46 // boost random number generator (0,1)
47 boost::random::uniform_int_distribution<> rands{0,1};
48
49 // progress bar
50 tula::logging::progressbar pb(
51 [&](const auto &msg) { logger->info("{}", msg); }, 100, "citlali progress ");
52
53 // grppi generator function. gets time chunk data from files sequentially and passes them to grppi::farm
54 grppi::pipeline(tula::grppi_utils::dyn_ex(parallel_policy),
55 [&]() -> std::optional<tuple_t> {
56 // variable to hold current scan
57 static int scan = 0;
58 // loop through scans
59 while (scan < telescope.scan_indices.cols()) {
60 // update progress bar
61 pb.count(telescope.scan_indices.cols(), 1);
62
63 // create rtcdata
64 TCData<TCDataKind::RTC, Eigen::MatrixXd> rtcdata;
65 // get scan indices
66 rtcdata.scan_indices.data = telescope.scan_indices.col(scan);
67 // current scan
68 rtcdata.index.data = scan;
69
70 // populate noise matrix (do outside of parallelized region for thread safety)
71 if (run_noise) {
72 if (omb.randomize_dets) {
73 // n_noise x n_dets
74 rtcdata.noise.data = Eigen::Matrix<int, Eigen::Dynamic, Eigen::Dynamic>::Zero(omb.n_noise, calib.n_dets)
75 .unaryExpr([&](int dummy){ return 2 * rands(eng) - 1; });
76 } else {
77 // n_noise
78 rtcdata.noise.data = Eigen::Matrix<int, Eigen::Dynamic, 1>::Zero(omb.n_noise)
79 .unaryExpr([&](int dummy){ return 2 * rands(eng) - 1; });
80 }
81 }
82 // vector to store kids data
83 std::vector<kids::KidsData<kids::KidsDataKind::RawTimeStream>> scan_rawobs;
84 // get kids data
85 if (!interp_over_gaps) {
86 scan_rawobs = kidsproc.load_rawobs(rawobs, scan, telescope.scan_indices, start_indices, end_indices);
87 }
88 else {
89 scan_rawobs = kidsproc.load_rawobs_2(rawobs, scan, telescope.scan_indices, start_indices, t_common, nw_times, 1 / (2 * telescope.fsmp));
90 }
91 // current length of outer scans
92 Eigen::Index sl = rtcdata.scan_indices.data(3) - rtcdata.scan_indices.data(2) + 1;
93
94 // get raw tod from files
95 if (!interp_over_gaps) {
96 rtcdata.scans.data = kidsproc.populate_rtc(scan_rawobs, sl, calib.n_dets, tod_type);
97 }
98 else {
99 rtcdata.scans.data = kidsproc.populate_rtc_2(scan_rawobs, t_common, nw_times, masks, scan, 1 / (2 * telescope.fsmp),
100 telescope.scan_indices, sl, calib.n_dets, tod_type);
101 }
102 // try and clear input vector
103 std::vector<kids::KidsData<kids::KidsDataKind::RawTimeStream>>().swap(scan_rawobs);
104
105 // increment scan
106 scan++;
107 // return rtcdata
108 return rtcdata;
109 }
110 // reset scan to zero for each obs
111 scan = 0;
112 return {};
113 },
114
115 // run the farm
116 run());
117
118 if (run_mapmaking) {
119 // normalize maps
120 logger->info("normalizing maps");
121 if (map_method != "maximum_likelihood") {
122 if (rtcproc.run_polarization) {
123 omb.normalize_polarized_maps();
124 }
125 else {
126 omb.normalize_maps();
127 }
128 }
129 // calculate map psds
130 logger->info("calculating map psd");
131 omb.calc_map_psd();
132 // calculate map histograms
133 logger->info("calculating map histogram");
134 omb.calc_map_hist();
135 // calculate mean error
136 omb.calc_median_err();
137 // calculate mean rms
138 omb.calc_median_rms();
139
140 // write map summary
141 if (verbose_mode) {
142 write_map_summary(omb);
143 }
144 }
145}

147auto Lali::run() {
148 using input_t = TCData<TCDataKind::RTC, Eigen::MatrixXd>;
149
150 auto farm = grppi::farm(n_threads,[&](input_t &rtcdata) {
151 // starting index for scan
152 Eigen::Index si = rtcdata.scan_indices.data(2);
153 // current length of outer scans
154 Eigen::Index sl = rtcdata.scan_indices.data(3) - rtcdata.scan_indices.data(2) + 1;
155
156 // copy scan's telescope vectors
157 for (const auto& x: telescope.tel_data) {
158 rtcdata.tel_data.data[x.first] = telescope.tel_data[x.first].segment(si,sl);
159 }
160
161 // copy pointing offsets
162 for (const auto& [axis,offset]: pointing_offsets_arcsec) {
163 rtcdata.pointing_offsets_arcsec.data[axis] = offset.segment(si,sl);
164 }
165
166 // get hwpr
167 if (rtcproc.run_polarization) {
168 if (calib.run_hwpr) {
169 rtcdata.hwpr_angle.data = calib.hwpr_angle.segment(si + hwpr_start_indices, sl);
170 }
171 }
172
173 // set up flags
174 rtcdata.flags.data.resize(rtcdata.scans.data.rows(), rtcdata.scans.data.cols());
175 rtcdata.flags.data.setConstant(0);
176
177 if (interp_over_gaps) {
178 int i = 0;
179 for (auto const& [key, val] : calib.nw_limits) {
180 auto& mask = masks[i];
181
182 Eigen::Index start = std::get<0>(calib.nw_limits[key]);
183 Eigen::Index end = std::get<1>(calib.nw_limits[key]) - 1;
184
185 for (int j = 0; j < rtcdata.flags.data.rows(); ++j) {
186 if (!mask(j)) {
187 rtcdata.flags.data.block(0, start, rtcdata.flags.data.rows(), end - start + 1).setOnes();
188 }
189 }
190 i++;
191 }
192 }
193
194 // create PTCData
195 TCData<TCDataKind::PTC,Eigen::MatrixXd> ptcdata;
196
197 logger->info("starting scan {}. {}/{} scans completed", rtcdata.index.data + 1, n_scans_done,
198 telescope.scan_indices.cols());
199
200 // run rtcproc
201 logger->info("raw time chunk processing for scan {}", rtcdata.index.data + 1);
202 auto map_indices = rtcproc.run(rtcdata, ptcdata, calib, telescope, omb.pixel_size_rad, map_grouping);
203
204 // remove flagged detectors
205 rtcproc.remove_flagged_dets(ptcdata, calib.apt);
206
207 // remove outliers before cleaning
208 auto calib_scan = rtcproc.remove_bad_dets(ptcdata, calib, map_grouping);
209
210 // remove duplicate tones
211 if (!telescope.sim_obs) {
212 calib_scan = rtcproc.remove_nearby_tones(ptcdata, calib, map_grouping);
213 }
214
215 // write rtc timestreams
216 if (run_tod_output && !tod_filename.empty()) {
217 if (tod_output_type == "rtc" || tod_output_type == "both") {
218 logger->info("writing raw time chunk");
219 rtcproc.append_to_netcdf(ptcdata, tod_filename["rtc"], map_grouping, telescope.pixel_axes,
220 ptcdata.pointing_offsets_arcsec.data, calib);
221 }
222 }
223
224 // if running fruit loops and a map has been read in
225 if (ptcproc.run_fruit_loops && !ptcproc.tod_mb.signal.empty()) {
226 logger->info("subtracting map from tod");
227 // subtract map
228 ptcproc.map_to_tod<timestream::TCProc::SourceType::NegativeMap>(ptcproc.tod_mb, ptcdata, calib,
229 map_indices, telescope.pixel_axes,
230 map_grouping);
231 }
232
233 // run cleaning
234 logger->info("processed time chunk processing for scan {}", ptcdata.index.data + 1);
235 ptcproc.run(ptcdata, ptcdata, calib, telescope.pixel_axes, map_grouping);
236
237 // if running fruit loops and a map has been read in
238 if (ptcproc.run_fruit_loops && !ptcproc.tod_mb.signal.empty()) {
239 if (run_mapmaking && run_noise) {
240 // calculate weights
241 logger->info("calculating weights");
242 ptcproc.calc_weights(ptcdata, calib.apt, telescope);
243
244 // reset weights to median
245 auto calib_scans = ptcproc.reset_weights(ptcdata, calib, map_grouping);
246
247 // populate noise maps only
248 bool run_omb = false;
249 logger->info("populating noise maps");
250 if (map_method=="naive") {
251 naive_mm.populate_maps_naive(ptcdata, omb, cmb, map_indices, telescope.pixel_axes,
252 calib.apt, telescope.d_fsmp, run_omb, run_noise);
253 }
254 else if (map_method=="jinc") {
255 jinc_mm.populate_maps_jinc(ptcdata, omb, cmb, map_indices, telescope.pixel_axes,
256 calib.apt, telescope.d_fsmp, run_omb, run_noise);
257 }
258 }
259 logger->info("adding map to tod");
260 // add map back
261 ptcproc.map_to_tod<timestream::TCProc::SourceType::Map>(ptcproc.tod_mb, ptcdata, calib,
262 map_indices, telescope.pixel_axes,
263 map_grouping);
264 }
265
266 // remove outliers after cleaning
267 calib_scan = ptcproc.remove_bad_dets(ptcdata, calib, map_grouping);
268
269 // calculate weights
270 logger->info("calculating weights");
271 ptcproc.calc_weights(ptcdata, calib.apt, telescope);
272
273 // reset weights to median
274 calib_scan = ptcproc.reset_weights(ptcdata, calib, map_grouping);
275
276 // write ptc timestreams
277 if (run_tod_output && !tod_filename.empty()) {
278 if (tod_output_type == "ptc" || tod_output_type == "both") {
279 logger->info("writing processed time chunk");
280 ptcproc.append_to_netcdf(ptcdata, tod_filename["ptc"], map_grouping, telescope.pixel_axes,
281 ptcdata.pointing_offsets_arcsec.data, calib);
282 }
283 }
284
285 // write out chunk summary
286 if (verbose_mode) {
287 write_chunk_summary(ptcdata);
288 }
289
290 // write stats
291 logger->debug("calculating stats");
292 diagnostics.calc_stats(ptcdata);
293
294 // populate maps
295 if (run_mapmaking) {
296 // make signal, weight, kernel, and coverage maps
297 bool run_omb = true;
298 bool run_noise_fruit = run_noise;
299
300 // if running fruit loops, noise maps are made on source
301 // subtracted timestreams so don't make them here
302 if (ptcproc.run_fruit_loops && !ptcproc.tod_mb.signal.empty()) {
303 run_noise_fruit = false;
304 }
305
306 // populate maps with current time chunk
307 logger->info("populating maps");
308 if (map_method=="naive") {
309 naive_mm.populate_maps_naive(ptcdata, omb, cmb, map_indices, telescope.pixel_axes,
310 calib.apt, telescope.d_fsmp, run_omb, run_noise_fruit);
311 }
312 else if (map_method=="jinc") {
313 jinc_mm.populate_maps_jinc(ptcdata, omb, cmb, map_indices, telescope.pixel_axes,
314 calib.apt, telescope.d_fsmp, run_omb, run_noise_fruit);
315 }
316 else if (map_method=="maximum_likelihood") {
317 ml_mm.populate_maps_ml(ptcdata, omb, cmb, map_indices, telescope.pixel_axes,
318 calib, telescope.d_fsmp, run_omb, run_noise_fruit);
319 }
320 }
321
322 // increment number of completed scans
323 n_scans_done++;
324 logger->info("done with scan {}. {}/{} scans completed", ptcdata.index.data + 1, n_scans_done, telescope.scan_indices.cols());
325 });
326
327 return farm;
328}

331void Lali::output() {
332 // pointer to map buffer
333 mapmaking::MapBuffer* mb = nullptr;
334 // pointer to data file fits vector
335 std::vector<fitsIO<file_type_enum::write_fits, CCfits::ExtHDU*>>* f_io = nullptr;
336 // pointer to noise file fits vector
337 std::vector<fitsIO<file_type_enum::write_fits, CCfits::ExtHDU*>>* n_io = nullptr;
338
339 // directory name
340 std::string dir_name;
341
342 // set common variables depending on map_type
343 if constexpr (map_type == mapmaking::RawObs || map_type == mapmaking::FilteredObs) {
344 mb = &omb;
345 dir_name = obsnum_dir_name + (map_type == mapmaking::RawObs ? "raw/" : "filtered/");
346 f_io = (map_type == mapmaking::RawObs) ? &fits_io_vec : &filtered_fits_io_vec;
347 n_io = (map_type == mapmaking::RawObs) ? &noise_fits_io_vec : &filtered_noise_fits_io_vec;
348
349 if constexpr (map_type == mapmaking::RawObs) {
350 // write stats file
351 write_stats();
352 if (run_tod_output && !tod_filename.empty()) {
353 // add tod header information
354 add_tod_header();
355 }
356 }
357 }
358 else if constexpr (map_type == mapmaking::RawCoadd || map_type == mapmaking::FilteredCoadd) {
359 mb = &cmb;
360 dir_name = coadd_dir_name + (map_type == mapmaking::RawCoadd ? "raw/" : "filtered/");
361 f_io = (map_type == mapmaking::RawCoadd) ? &coadd_fits_io_vec : &filtered_coadd_fits_io_vec;
362 n_io = (map_type == mapmaking::RawCoadd) ? &coadd_noise_fits_io_vec : &filtered_coadd_noise_fits_io_vec;
363 }
364
365 if (run_mapmaking) {
366 // wiener filtered maps write before this and are deleted from the vector.
367 if (!f_io->empty()) {
368 {
369 // progress bar
370 tula::logging::progressbar pb(
371 [&](const auto &msg) { logger->info("{}", msg); }, 100, "output progress ");
372
373 for (Eigen::Index i=0; i<f_io->size(); ++i) {
374 // get the array for the given map
375 // add primary hdu
376 logger->debug("adding primary header to file {}",i);
377 add_phdu(f_io, mb, i);
378
379 // add primary hdu to noise maps
380 if (!mb->noise.empty()) {
381 logger->debug("adding primary header to noise file {}",i);
382 add_phdu(n_io, mb, i);
383 }
384 }
385
386 logger->debug("done adding primary headers");
387
388 // write the maps
389 for (Eigen::Index i=0; i<n_maps; ++i) {
390 // update progress bar
391 pb.count(n_maps, 1);
392 write_maps(f_io,n_io,mb,i);
393 }
394 }
395
396 logger->info("maps have been written to:");
397 for (Eigen::Index i=0; i<f_io->size(); ++i) {
398 logger->info("{}.fits",f_io->at(i).filepath);
399 }
400 }
401
402 // clear fits file vectors to ensure its closed.
403 std::vector<fitsIO<file_type_enum::write_fits, CCfits::ExtHDU*>>().swap(*f_io);
404 std::vector<fitsIO<file_type_enum::write_fits, CCfits::ExtHDU*>>().swap(*n_io);
405
406 // write psd and histogram files
407 logger->debug("writing psds");
408 write_psd<map_type>(mb, dir_name);
409 logger->debug("writing histograms");
410 write_hist<map_type>(mb, dir_name);
411
412 // write source table
413 if (run_source_finder) {
414 logger->debug("writing source table");
415 write_sources<map_type>(mb, dir_name);
416 }
417 }
